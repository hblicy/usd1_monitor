```python
# usd1_monitor/rpc.py
class RpcError(RuntimeError):
    pass


class JsonRpcClient:
    def __init__(self, urls: list[str], http) -> None:
        self._urls = urls
        self._http = http
        self._request_id = 0

    async def call(self, method: str, params: list) -> object:
        failures: list[str] = []
        for url in self._urls:
            self._request_id += 1
            payload = {"jsonrpc": "2.0", "id": self._request_id, "method": method, "params": params}
            try:
                body = await self._http.post_json(url, payload)
                if "error" in body:
                    error = body["error"]
                    raise RpcError(f"RPC {method} failed: {error.get('code')} {error.get('message')}")
                if "result" not in body:
                    raise RpcError(f"RPC {method} response has no result")
                return body["result"]
            except Exception as exc:
                failures.append(f"{type(exc).__name__}: {exc}")
        raise RpcError(f"all RPC endpoints failed for {method}: {' | '.join(failures)}")
```

Ensure the HTTP layer sanitizes URL query strings before the exception reaches logs.

```python
# append to tests/fakes.py
class FakeRpc:
    def __init__(self) -> None:
        self.responses: dict[str, list[object]] = {}
        self.calls: list[tuple[str, list]] = []

    def result(self, method: str, value: object) -> None:
        self.responses.setdefault(method, []).append(value)

    async def call(self, method: str, params: list) -> object:
        self.calls.append((method, params))
        value = self.responses[method].pop(0)
        if isinstance(value, Exception):
            raise value
        return value

    def calls_for(self, method: str) -> list[list]:
        return [params for called_method, params in self.calls if called_method == method]
```

```python
# public range calculation in usd1_monitor/collectors/evm.py
def scan_range(latest: int, confirmation_depth: int, cursor: int | None, overlap: int) -> tuple[int, int] | None:
    safe_head = latest - confirmation_depth
    if safe_head < 0:
        return None
    start = max(0, (cursor + 1 - overlap) if cursor is not None else safe_head - overlap + 1)
    return None if start > safe_head else (start, safe_head)
```
